cmake_minimum_required(VERSION 3.16)
project(audio_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(TO_CMAKE_PATH "$ENV{IDF_PATH}" IDF_ROOT)
add_library(cjson ${IDF_ROOT}/components/json/cJSON/cJSON.c)
target_include_directories(cjson PUBLIC ${IDF_ROOT}/components/json/cJSON)
add_executable(mcp_test mcp_test.cpp ${ROOT}/firmware/main/local_mcp.cpp)
target_include_directories(mcp_test PRIVATE ${ROOT}/firmware/main)
target_link_libraries(mcp_test PRIVATE cjson)
target_compile_options(mcp_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(audio_priority_test audio_priority_test.cpp ${ROOT}/firmware/components/shared_audio/shared_audio.cpp)
target_include_directories(audio_priority_test PRIVATE stubs ${ROOT}/firmware/components/shared_audio)
target_compile_options(audio_priority_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME mcp COMMAND mcp_test ${ROOT}/firmware/main/mcp_tools.json)
add_test(NAME audio_priority COMMAND audio_priority_test)
add_executable(http_helpers_test http_helpers_test.cpp)
target_include_directories(http_helpers_test PRIVATE stubs ${ROOT}/firmware/main)
target_link_libraries(http_helpers_test PRIVATE cjson)
target_compile_options(http_helpers_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME http_helpers COMMAND http_helpers_test)
add_executable(xiaozhi_protocol_test xiaozhi_protocol_test.cpp)
target_include_directories(xiaozhi_protocol_test PRIVATE ${ROOT}/firmware/main)
target_compile_options(xiaozhi_protocol_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME xiaozhi_protocol COMMAND xiaozhi_protocol_test)
add_executable(boot_control_test boot_control_test.cpp)
target_include_directories(boot_control_test PRIVATE ${ROOT}/firmware/main)
target_compile_options(boot_control_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME boot_control COMMAND boot_control_test)
